# realpath: one session, with stdout on the terminal and into a pipe, at 40, 80 and 120 columns.
# Rows starting with "! " went to stderr, and "? N" is the exit status. Where the variants
# differ, each output is under the variants it belongs to. See README.md.

$ realpath . /bin/ls /home/user/../user/README.md
/home/guest
/usr/bin/ls
/home/guest/README.md
? 0
$ realpath -m drafts/new.txt
/home/guest/drafts/new.txt
? 0
$ realpath --relative-to=documents projects
../projects
? 0
$ realpath -e nope
! realpath: nope: No such file or directory
? 1
$ realpath nope/x
! realpath: nope/x: No such file or directory
? 1
