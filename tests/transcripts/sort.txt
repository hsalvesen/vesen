# sort: one session, with stdout on the terminal and into a pipe, at 40, 80 and 120 columns.
# Rows starting with "! " went to stderr, and "? N" is the exit status. Where the variants
# differ, each output is under the variants it belongs to. See README.md.

$ sort /etc/shells
/bin/sh
/bin/vesh
# /etc/shells: valid login shells
? 0
$ cut -d : -f 7 /etc/passwd | sort | uniq -c
      3 /bin/vesh
      1 /usr/sbin/nologin
? 0
$ sort -t : -k 3,3n /etc/passwd
root:x:0:0:root:/root:/bin/vesh
guest:x:1000:1000:Guest:/home/guest:/bin/vesh
has:x:1001:1001:Has Salvesen:/home/has:/bin/vesh
nobody:x:65534:65534:nobody:/nonexistent:/usr/sbin/nologin
? 0
$ printf 'b\na\n' | sort -c
! sort: -:2: disorder: a
? 1
